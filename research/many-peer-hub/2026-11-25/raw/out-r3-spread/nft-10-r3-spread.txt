table inet infraternal_mesh {
	comment "infraternal.mesh.v1"
	set mesh4 {
		type ipv4_addr
		elements = { 10.254.0.1, 10.254.1.1,
			     10.254.1.2, 10.254.1.3,
			     10.254.1.4, 10.254.1.5,
			     10.254.1.6, 10.254.1.7,
			     10.254.1.8, 10.254.1.9,
			     10.254.1.10 }
	}

	set mesh6 {
		type ipv6_addr
		elements = { fd01::1,
			     fd01::1:1,
			     fd01::1:2,
			     fd01::1:3,
			     fd01::1:4,
			     fd01::1:5,
			     fd01::1:6,
			     fd01::1:7,
			     fd01::1:8,
			     fd01::1:9,
			     fd01::1:a }
	}

	chain ingress {
		type filter hook prerouting priority raw - 10; policy accept;
		iifname "lo" ip saddr 10.254.0.1 accept
		iifname "lo" ip6 saddr fd01::1 accept
		iifname != "infraternal0" ip saddr @mesh4 drop
		iifname != "infraternal0" ip6 saddr @mesh6 drop
		ip daddr 10.254.0.1 iifname != { "lo", "infraternal0" } drop
		ip6 daddr fd01::1 iifname != { "lo", "infraternal0" } drop
	}

	chain forward {
		type filter hook forward priority filter - 10; policy accept;
		ip daddr 10.254.0.1 iifname != { "lo", "infraternal0" } drop
		ip6 daddr fd01::1 iifname != { "lo", "infraternal0" } drop
		iifname "infraternal0" drop
		oifname "infraternal0" drop
	}

	chain input {
		type filter hook input priority filter - 10; policy accept;
		ip daddr 10.254.0.1 iifname != { "lo", "infraternal0" } drop
		ip6 daddr fd01::1 iifname != { "lo", "infraternal0" } drop
	}
}
